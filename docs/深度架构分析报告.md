## 三、窗口截图实现方式

### 3.1 当前实现

```python
def capture_window(hwnd, save_path=None):
    """
    使用 BitBlt 方式截图
    
    实现步骤：
    1. GetWindowDC 获取窗口 DC
    2. CreateCompatibleDC 创建兼容 DC
    3. BitBlt 拷贝像素数据
    4. PIL.Image.frombuffer 转换格式
    """
    hwndDC = win32gui.GetWindowDC(hwnd)
    mfcDC = win32ui.CreateDCFromHandle(hwndDC)
    saveDC = mfcDC.CreateCompatibleDC()
    
    saveBitMap = win32ui.CreateBitmap()
    saveBitMap.CreateCompatibleBitmap(mfcDC, w, h)
    saveDC.SelectObject(saveBitMap)
    
    # 核心：使用 BitBlt 拷贝
    saveDC.BitBlt((0, 0), (w, h), mfcDC, (offset_x, offset_y), win32con.SRCCOPY)
    
    # 转换为 PIL Image
    bmpstr = saveBitMap.GetBitmapBits(True)
    im = Image.frombuffer('RGB', (w, h), bmpstr, 'raw', 'BGRX', 0, 1)
```

### 3.2 截图方式对比

| 方式 | 性能 | 兼容性 | 后台支持 | 适用场景 |
|------|------|--------|----------|----------|
| **BitBlt (当前)** | ⭐⭐⭐⭐ | ⭐⭐⭐⭐ | ⭐⭐⭐ | 窗口模式 |
| PrintWindow | ⭐⭐⭐ | ⭐⭐⭐ | ⭐⭐⭐⭐ | 后台截图 |
| DXGI Desktop Duplication | ⭐⭐⭐⭐⭐ | ⭐⭐ | ⭐⭐⭐⭐⭐ | 全屏/高性能 |
| WinRT Capture | ⭐⭐⭐⭐⭐ | ⭐⭐ | ⭐⭐⭐⭐⭐ | Windows 10+ |

### 3.3 当前方案评估

**优点：**
- 实现简单，兼容性好
- 支持部分后台截图（窗口被遮挡时）

**缺点：**
- 窗口最小化时无法截图
- 需要正确处理客户区偏移
- 每次截图都有 GDI 对象创建/销毁开销

### 4.3 smart_sleep 实现

```python
def smart_sleep(self, seconds: float) -> None:
    """
    智能睡眠 - 可中断的等待
    
    原理：
    1. 将长时间睡眠切分为 100ms 时间片
    2. 每个时间片后检查暂停/停止信号
    3. 使用 time.sleep 让出 CPU
    """
    time_slice = 0.1  # 100ms
    
    while True:
        remaining = end_time - time.time()
        if remaining <= 0:
            break
        
        sleep_duration = min(time_slice, remaining)
        time.sleep(sleep_duration)
        self.check_status()  # 检查控制信号
```

### 7.2 任务系统设计优化

#### 问题：任务注册分散，扩展困难

**建议：装饰器 + 注册表模式**

```python
# task_registry.py
class TaskRegistry:
    """任务注册表"""
    _tasks = {}
    _subtasks = {}
    
    @classmethod
    def register(cls, name: str, category: str = 'daily'):
        """任务注册装饰器"""
        def decorator(func):
            cls._tasks[name] = {
                'func': func,
                'category': category,
                'config_def': getattr(func, '_config_def', None)
            }
            return func
        return decorator
    
    @classmethod
    def subtask(cls, parent: str, name: str):
        """子任务注册装饰器"""
        def decorator(func):
            if parent not in cls._subtasks:
                cls._subtasks[parent] = {}
            cls._subtasks[parent][name] = func
            return func
        return decorator
    
    @classmethod
    def get_task(cls, name: str):
        return cls._tasks.get(name)
    
    @classmethod
    def get_all_tasks(cls, category: str = None):
        if category:
            return {k: v for k, v in cls._tasks.items() 
                    if v['category'] == category}
        return cls._tasks.copy()
```

### 7.3 图像识别与截图性能优化

#### 问题1：频繁截图

**解决方案：帧缓存机制**

```python
class FrameCache:
    """
    帧缓存管理器
    
    在同一帧内复用截图，避免重复捕获
    """
    def __init__(self, ttl: float = 0.1):
        self._cache = {}
        self._ttl = ttl
        self._lock = threading.Lock()
    
    def get_frame(self, hwnd: int) -> Optional[Image.Image]:
        """获取帧（带缓存）"""
        cache_key = (hwnd, int(time.time() / self._ttl))
        
        with self._lock:
            if cache_key not in self._cache:
                self._cache[cache_key] = capture_window(hwnd)
                # 清理旧缓存
                self._cleanup_old(cache_key)
            
            return self._cache[cache_key]
    
    def invalidate(self, hwnd: int = None):
        """使缓存失效"""
        with self._lock:
            if hwnd:
                self._cache = {k: v for k, v in self._cache.items() 
                              if k[0] != hwnd}
            else:
                self._cache.clear()
```

```python
            # ... 其他常用模板
        ]
        for path in common_templates:
            self.load(path)
    
    def load(self, path: str) -> np.ndarray:
        """加载模板（带缓存）"""
        if path in self._cache:
            self._cache.move_to_end(path)
            return self._cache[path]
        
        template = cv2.imread(path)
        if template is not None:
            self._cache[path] = template
            if len(self._cache) > self._max_size:
                self._cache.popitem(last=False)
        
        return template
```

```python
class SmartCapture:
    """
    智能截图器
    
    根据窗口状态自动选择最佳截图方式
    """
    def __init__(self):
        self._methods = [
            ('bitblt', self._capture_bitblt),
            ('print_window', self._capture_print_window),
        ]
        self._best_method = {}
    
    def capture(self, hwnd: int) -> Optional[Image.Image]:
        """智能截图"""
        # 检查窗口是否最小化
        if self._is_minimized(hwnd):
            return self._capture_print_window(hwnd)
        
        # 使用缓存的最佳方式
        if hwnd in self._best_method:
            return self._best_method[hwnd](hwnd)
        
        # 测试选择最佳方式
        for name, method in self._methods:
            result = method(hwnd)
            if result is not None:
                self._best_method[hwnd] = method
                return result
        
        return None
```

```python
class ActionExecutor:
    """
    动作执行器
    
    提供带确认的动作执行
    """
    def click_with_confirm(self, hwnd: int, pos: Tuple[int, int],
                           confirm_template: str = None,
                           confirm_disappear: str = None,
                           max_retries: int = 3,
                           confirm_timeout: float = 5.0) -> bool:
        """
        带确认的点击
        
        点击后验证是否生效：
        1. confirm_template: 期望出现的模板
        2. confirm_disappear: 期望消失的模板
        """
        for attempt in range(max_retries):
            background_click(hwnd, pos[0], pos[1])
            
            start_time = time.time()
            while time.time() - start_time < confirm_timeout:
                task_controller.smart_sleep(0.5)
                
                if confirm_template:
                    if find_image(hwnd, confirm_template):
                        return True
                
                if confirm_disappear:
                    if not find_image(hwnd, confirm_disappear):
                        return True
            
            logger.warning(f"点击确认失败，重试 {attempt + 1}/{max_retries}")
        
        return False
    
    def wait_for_animation(self, hwnd: int, 
                           stability_threshold: int = 3,
                           check_interval: float = 0.5) -> bool:
        """
        等待动画结束
        
        通过连续帧对比检测动画是否结束
        """
        stable_count = 0
        last_frame = None
        
        while stable_count < stability_threshold:
            current_frame = capture_window(hwnd)
            
            if last_frame is not None:
                diff = self._frame_difference(last_frame, current_frame)
                if diff < 0.01:  # 差异小于 1%
                    stable_count += 1
                else:
                    stable_count = 0
            
            last_frame = current_frame
            task_controller.smart_sleep(check_interval)
        
        return True
```

```python
class GameStateSnapshot:
    """
    游戏状态快照
    
    记录任务执行前的状态，用于异常恢复
    """
    def __init__(self, hwnd: int):
        self.hwnd = hwnd
        self.timestamp = time.time()
        self.open_windows = self._detect_open_windows()
        self.location = self._detect_location()
    
    def restore(self):
        """恢复到快照状态"""
        # 关闭所有意外窗口
        for window in self.open_windows:
            close_window(window)
        
        # 返回原位置
        if self.location:
            teleport_to(self.location)

class TaskExecutor:
    """带恢复机制的任务执行器"""
    
    def execute_with_recovery(self, task_func, hwnd, max_retries=3):
        """带恢复机制执行任务"""
        for attempt in range(max_retries):
            snapshot = GameStateSnapshot(hwnd)
            
            try:
                result = task_func(hwnd)
                return result
            
            except TaskStuckException:
                logger.warning(f"任务卡死，尝试恢复 ({attempt + 1}/{max_retries})")
                snapshot.restore()
            
            except GameCrashException:
                logger.error("游戏崩溃，停止执行")
                raise
        
        return False
```

```python
class TaskPlugin(ABC):
    """任务插件基类"""
    
    @property
    @abstractmethod
    def name(self) -> str:
        """插件名称"""
        pass
    
    @property
    @abstractmethod
    def version(self) -> str:
        """插件版本"""
        pass
    
    @abstractmethod
    def execute(self, context: TaskContext) -> TaskResult:
        """执行任务"""
        pass
    
    def on_load(self):
        """插件加载时调用"""
        pass
    
    def on_unload(self):
        """插件卸载时调用"""
        pass


class PluginManager:
    """插件管理器"""
    
    def __init__(self, plugin_dir: str):
        self._plugins = {}
        self._plugin_dir = plugin_dir
    
    def load_plugin(self, plugin_path: str):
        """加载插件"""
        spec = importlib.util.spec_from_file_location("plugin", plugin_path)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        
        for attr in dir(module):
            obj = getattr(module, attr)
            if isinstance(obj, type) and issubclass(obj, TaskPlugin):
                plugin = obj()
                self._plugins[plugin.name] = plugin
                plugin.on_load()
    
    def get_plugin(self, name: str) -> Optional[TaskPlugin]:
        return self._plugins.get(name)
```
