## 一、按钮与运行状态管理 (Button & Run State)

### 1.1 ButtonState 类定义

```python
class ButtonState:
    """
    按钮状态枚举类
    
    定义按钮的四种状态：
        - IDLE: 初始状态，主控按钮"开始执行"，暂停按钮禁用
        - RUNNING: 运行状态，主控按钮"强制停止"，暂停按钮"暂停运行"
        - PAUSED: 暂停状态，主控按钮"强制停止"，暂停按钮"继续运行"
        - STOPPING: 停止过渡状态，两个按钮都禁用
    """
    IDLE = "idle"
    RUNNING = "running"
    PAUSED = "paused"
    STOPPING = "stopping"
```

### 1.3 _set_button_state 方法完整实现

```python
def _set_button_state(self, state: str):
    """
    设置按钮状态
    
    根据状态更新按钮的文字、样式和启用状态
    
    参数：
        state: 目标状态（ButtonState 枚举值）
    """
    self._button_state = state
    
    if state == ButtonState.IDLE:
        self.start_btn.setText("开始执行")
        self.start_btn.setEnabled(True)
        self.start_btn.setStyleSheet("")
        self.stop_btn.setText("暂停运行")
        self.stop_btn.setEnabled(False)
        self.stop_btn.setStyleSheet("")
        self.pick_btn.set_disabled(False)
        self.unbind_btn.set_running(False)
        
    elif state == ButtonState.RUNNING:
        self.start_btn.setText("强制停止")
        self.start_btn.setEnabled(True)
        self.start_btn.setStyleSheet(
            f"background-color: {self._colors['danger']}; color: white;"
        )
        self.stop_btn.setText("暂停运行")
        self.stop_btn.setEnabled(True)
        self.stop_btn.setStyleSheet("")
        self.pick_btn.set_disabled(True)
        self.unbind_btn.set_running(True)
        
    elif state == ButtonState.PAUSED:
        self.start_btn.setText("强制停止")
        self.start_btn.setEnabled(True)
        self.start_btn.setStyleSheet(
            f"background-color: {self._colors['danger']}; color: white;"
        )
        self.stop_btn.setText("继续运行")
        self.stop_btn.setEnabled(True)
        self.stop_btn.setStyleSheet(
            f"background-color: {self._colors['success']}; color: white;"
        )
        self.pick_btn.set_disabled(True)
        self.unbind_btn.set_running(True)
        
    elif state == ButtonState.STOPPING:
        self.start_btn.setText("正在停止...")
        self.start_btn.setEnabled(False)
        self.start_btn.setStyleSheet(
            f"background-color: {self._colors['secondary']}; color: white;"
        )
        self.stop_btn.setText("暂停运行")
        self.stop_btn.setEnabled(False)
        self.stop_btn.setStyleSheet("")
        self.pick_btn.set_disabled(True)
```

### 1.4 状态转换相关方法

```python
def _on_pause_btn_clicked(self):
    """
    暂停按钮点击处理
    
    根据当前状态切换暂停/继续：
        - RUNNING -> PAUSED: 暂停任务
        - PAUSED -> RUNNING: 继续任务
    """
    if self._button_state == ButtonState.RUNNING:
        task_controller.pause()
        self._set_button_state(ButtonState.PAUSED)
        self.logger.info("任务已暂停")
    elif self._button_state == ButtonState.PAUSED:
        task_controller.resume()
        self._set_button_state(ButtonState.RUNNING)
        self.logger.info("任务已继续")

def _toggle_run_state(self):
    """
    切换运行状态
    
    当任务正在运行时，点击主控按钮会触发停止操作。
    停止时禁用启动按钮，等待线程结束后恢复。
    """
    if self.worker and self.worker.isRunning():
        self._set_button_state(ButtonState.STOPPING)
        self.worker.stop()
        self.logger.warning("正在停止运行...")

def _on_task_finished(self):
    """
    任务完成回调
    
    当后台线程完成时恢复按钮状态到初始状态
    """
    self._set_button_state(ButtonState.IDLE)
```

    self.logger.success(f"成功绑定窗口，句柄: {hwnd}")

### 2.3 _unbind_window 方法（解绑窗口）

```python
def _unbind_window(self):
    """
    解绑窗口
    
    流程：
        1. 检查是否有运行中的脚本，如有则停止
        2. 等待脚本线程退出
        3. 释放输入资源
        4. 重置任务控制器状态
        5. 清除窗口句柄和更新UI
    """
    if self.worker and self.worker.isRunning():
        self.logger.warning("正在停止运行...")
        task_controller.stop()
        
        if not self.worker.wait(500):
            self.logger.warning("线程退出超时，强制终止")
            self.worker.terminate()
            self.worker.wait()
        
        release_tracked_inputs(self.bound_hwnd)
        task_controller.reset_all_events()
    
    self.bound_hwnd = None
    self.hwnd_label.setText("未绑定窗口")
    self.hwnd_label.setStyleSheet(
        f"font-weight: bold; color: {self._colors['text_secondary']}; font-size: 9pt; "
        f"border: 1px solid {self._colors['border_strong']}; "
        f"background-color: {self._colors['surface_elevated']};"
    )
    self.preview_label.clear()
    self.preview_label.setText("未绑定角色")
    self.pick_btn.set_bound_state(False)
    self.pick_btn.set_disabled(False)
    self.unbind_btn.set_enabled_state(False)
    self.unbind_btn.set_running(False)
    self.logger.info("已解除窗口绑定")
```

## 三、任务配置参数的流转 (Task Configuration Data)

### 3.1 _get_task_params_for_execution 方法

```python
def _get_task_params_for_execution(self) -> dict:
    """
    获取用于执行的任务参数（扁平化）
    
    使用 get_flattened_task_params 获取扁平化参数，
    确保传递给任务函数的参数是扁平化字典（全量提取）
    
    返回：
        dict: 任务参数字典（已映射）
    """
    params = {}
    for task_name in config.daily_tasks:
        if not config.has_task_config(task_name):
            continue
        
        flattened = self.task_config_panel.get_flattened_task_params(task_name)
        
        params[task_name] = {}
        for param_name, value in flattened.items():
            mapped_value = config.get_task_mapped_param(task_name, param_name, value)
            params[task_name][param_name] = mapped_value
    
    return params
```

### 3.3 参数流转调用链

```python
# 1. 用户点击"开始执行"
def _run_script(self):
    # ...
    task_params = self._get_task_params_for_execution()  # 收集参数
    self.worker = ScriptWorker(selected, self.bound_hwnd, task_params)  # 传递给 Worker
    self.worker.start()

# 2. Worker 线程执行
def run(self):
    for task_name in self.tasks:
        task_func = get_task(task_name)
        params = self.task_params.get(task_name, {})
        
        # 特殊任务参数处理
        if task_name == "摇钱树":
            choice = params.get("choice", 1)
            task_func(self.hwnd, choice=choice)
        elif task_name == "每日可换":
            result = task_func(self.hwnd, checkbox_state=params)
        else:
            result = task_func(self.hwnd)
```

## 四、解耦建议

### 4.1 状态管理器 (StateManager)

**建议提取的职责：**
- 按钮状态管理（ButtonState）
- 窗口绑定状态管理（bound_hwnd）
- 运行状态切换逻辑

**建议结构：**
```python
# src/core/state_manager.py
class StateManager:
    """统一状态管理器"""
    
    # 窗口状态
    bound_hwnd: Optional[int] = None
    
    # 运行状态
    button_state: str = ButtonState.IDLE
    
    # 信号
    window_bound = pyqtSignal(int)
    window_unbound = pyqtSignal()
    state_changed = pyqtSignal(str)
```

### 4.2 配置管理器 (ConfigManager)

**建议提取的职责：**
- 任务参数收集
- 参数映射转换
- 配置保存/加载

**建议结构：**
```python
# src/core/config_manager.py
class ConfigManager:
    """任务配置管理器"""
    
    def get_task_params(self) -> dict:
        """获取所有任务参数"""
        
    def save_config(self, name: str):
        """保存配置方案"""
        
    def load_config(self, name: str):
        """加载配置方案"""
```
